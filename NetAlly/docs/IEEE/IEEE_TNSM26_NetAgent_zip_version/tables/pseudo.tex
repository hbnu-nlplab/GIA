\begin{algorithm*}[t]
\caption{NetAgent: Multi-Agent Debate for Network Configuration QA}
\label{alg:netagent}
\begin{algorithmic}[1]

\Require Question $q$, raw context $C_{raw}$, answer type $\tau$,
         $T_{out}{=}3$, $T_{in}{=}3$
\Ensure Final answer $a^*$

\Statex \hrulefill
\Statex \textbf{Phase 1: Information Extracting \& Initial Reasoning}
\Statex \hrulefill

\State $feedback \leftarrow \emptyset$,\quad $t_{out} \leftarrow 0$

\Repeat \hfill\Comment{Outer Loop}
    \State $t_{out} \leftarrow t_{out} + 1$

    \State \textit{// Agent 1: Collector (Model B)}
    \State $C_{ext} \leftarrow \Call{Collector}{q,\; C_{raw},\; feedback}$
    \hfill\Comment{Extract relevant config blocks; generate from knowledge if $C_{raw}{=}\emptyset$}

    \State \textit{// Agent 2: Verifier (Model A)}
    \If{$\text{level}(q) \in \{\text{L4},\, \text{L5}\}$}
        \State $P \leftarrow C_{ext}$
        \hfill\Comment{BYPASS: preserve full context}
    \Else
        \State $P \leftarrow \Call{Verifier}{q,\; C_{ext}}$
        \hfill\Comment{Filter noise; produce high-purity passage}
    \EndIf

    \State \textit{// Agent 3: Synthesizer (Model A)}
    \If{$\text{level}(q) = \text{L5}$}
        \State $a_{cand} \leftarrow \Call{Synthesizer}{q,\; P,\; \tau,\;\text{6-step-fault-analysis}}$
    \Else
        \State $a_{cand} \leftarrow \Call{Synthesizer}{q,\; P,\; \tau}$
    \EndIf

\Statex \hrulefill
\Statex \quad\textbf{Phase 2: Adversarial Debate}
\Statex \hrulefill

    \State $t_{in} \leftarrow 0$

    \Repeat \hfill\Comment{Inner Loop}
        \State $t_{in} \leftarrow t_{in} + 1$

        \State \textit{// Agent 4: Proponent (Model A)}
        \State $\mathit{arg}_{pro} \leftarrow \Call{Proponent}{q,\; P,\; a_{cand}}$
        \hfill\Comment{Build supporting argument for $a_{cand}$}

        \State \textit{// Agent 5: Critic (Model B)}
        \State $(status,\; critique) \leftarrow \Call{Critic}{q,\; P,\; a_{cand},\; \mathit{arg}_{pro}}$

        \If{$status = \textsc{Accept}$}
            \State \Return $a_{cand}$
            \hfill\Comment{Consensus reached}
        \ElsIf{$status = \textsc{Continue\_Debate}$}
            \State $a_{cand} \leftarrow \Call{Proponent}{q,\; P,\; a_{cand},\; critique}$
            \hfill\Comment{Refine answer with critique}
        \ElsIf{$status = \textsc{Need\_More\_Info}$}
            \State $feedback \leftarrow critique$;\quad \textbf{break}
            \hfill\Comment{Backtrack to Outer Loop}
        \EndIf

    \Until{$status = \textsc{Accept}$ \textbf{ or } $t_{in} \geq T_{in}$}

    \If{$t_{in} \geq T_{in}$ \textbf{ and } $status \neq \textsc{Accept}$
        \textbf{ and } $status \neq \textsc{Need\_More\_Info}$}
        \State \Return $a_{cand}$
        \hfill\Comment{Inner Loop limit reached}
    \EndIf

\Until{$status = \textsc{Accept}$ \textbf{ or } $t_{out} \geq T_{out}$}

\State \Return $a_{cand}$
\hfill\Comment{Outer Loop limit reached}

\end{algorithmic}
\end{algorithm*}
